Cómo integrar en tu sitio

1. Agrega este código HTML donde quieras mostrar el QR:

<div id="whatsapp-qr-container"> <div id="whatsapp-status"></div> <img id="whatsapp-qr" style="display: none"> </div>

2. Agrega este código JavaScript:

<script> let whatsappSessionId = null; async function initWhatsAppQR() { try { const response = await fetch('http://localhost:3001/api/qr-status'); const data = await response.json(); if (data.success) { whatsappSessionId = data.sessionId; const qrImage = document.getElementById('whatsapp-qr'); const statusDiv = document.getElementById('whatsapp-status'); qrImage.src = data.qrDataUrl; qrImage.style.display = 'block'; statusDiv.textContent = 'Escanea el código QR con WhatsApp'; // Iniciar polling del estado checkConnectionStatus(); } } catch (error) { console.error('Error:', error); } } async function checkConnectionStatus() { if (!whatsappSessionId) return; try { const response = await fetch(`http://localhost:3001/api/session/${whatsappSessionId}/status-json`); const data = await response.json(); const statusDiv = document.getElementById('whatsapp-status'); const qrImage = document.getElementById('whatsapp-qr'); if (data.success) { if (data.isConnected) { statusDiv.textContent = '¡Conectado a WhatsApp!'; statusDiv.className = 'connected'; qrImage.style.display = 'none'; } else { setTimeout(checkConnectionStatus, 2000); } } } catch (error) { console.error('Error:', error); } } // Iniciar el proceso cuando la página cargue document.addEventListener('DOMContentLoaded', initWhatsAppQR); </script>

3. Asegúrate de que la URL del servidor (http://localhost:3001) coincida con la ubicación de tu servidor.
